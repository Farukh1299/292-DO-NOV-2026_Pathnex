Jenkinsfile - Build and Deploy to Kubernetes
pipeline {
 agent any
 environment {
     IMAGE_NAME = 'pathne-wen-app'
}
stages {
     stage('Build Docker Image') {
      steps {
        dokcer.build ("${env.IMAGE_NAME")
         }
     }
  }
stage('Push Dokcer Image') {
   steps {
     script {
      docker.withRegistry('https://dokcer.io', 'docker-credentials') {
      docker.image("${env.IMAGE_NAME}").push('latest')
          }
       }
     }
   }
stage ('Deploy to Kubernetes') {
     steps {
        script {
            sh 'kubectl apply -f deployment.yaml' 
           }
	}
      }	
    }
